include(FetchContent)

function(add_foliage_manifold)
  set(BUILD_SHARED_LIBS OFF)
  set(MANIFOLD_TEST OFF CACHE BOOL "" FORCE)
  set(MANIFOLD_CROSS_SECTION OFF CACHE BOOL "" FORCE)
  set(MANIFOLD_PAR OFF CACHE BOOL "" FORCE)
  set(MANIFOLD_DOWNLOADS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(manifold URL https://codeload.github.com/elalish/manifold/tar.gz/refs/tags/v3.5.3 URL_HASH SHA256=ebdbd7276f2df34690fb7d17d239f265446570152d712f5c951b7729eaa1f72f)
  FetchContent_MakeAvailable(manifold)
  set(FOLIAGE_MANIFOLD_SOURCE_DIR "${manifold_SOURCE_DIR}" PARENT_SCOPE)
endfunction()
add_foliage_manifold()
option(FOLIAGE_ALEMBIC "Build native Ogawa Alembic growth export" ON)
if(FOLIAGE_ALEMBIC)
  if(CMAKE_VERSION VERSION_LESS 3.29)
    message(FATAL_ERROR "Alembic requires CMake 3.29 or newer; use -DFOLIAGE_ALEMBIC=OFF for mesh and stage exports only")
  endif()
  function(add_foliage_alembic)
    set(BUILD_SHARED_LIBS OFF)
    set(IMATH_INSTALL_PKG_CONFIG OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(imath URL https://codeload.github.com/AcademySoftwareFoundation/Imath/tar.gz/refs/tags/v3.2.1 URL_HASH SHA256=b2c8a44c3e4695b74e9644c76f5f5480767355c6f98cde58ba0e82b4ad8c63ce)
    FetchContent_MakeAvailable(imath)
    target_include_directories(Imath INTERFACE $<BUILD_INTERFACE:${imath_SOURCE_DIR}/src>)
    set(ALEMBIC_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(USE_BINARIES OFF CACHE BOOL "" FORCE)
    set(USE_TESTS OFF CACHE BOOL "" FORCE)
    set(USE_HDF5 OFF CACHE BOOL "" FORCE)
    set(USE_PYALEMBIC OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(alembic URL https://codeload.github.com/alembic/alembic/tar.gz/refs/tags/1.8.12 URL_HASH SHA256=b6d916c40446e8c502c84273092ab3d98f3f7f6094f8a2b8203d23e2f1d2a4a0)
    FetchContent_MakeAvailable(alembic)
    set_property(TARGET Alembic Imath PROPERTY SYSTEM TRUE)
    install(FILES "${alembic_SOURCE_DIR}/LICENSE.txt" DESTINATION share/foliageutil/licenses RENAME alembic.txt)
    install(FILES "${imath_SOURCE_DIR}/LICENSE.md" DESTINATION share/foliageutil/licenses RENAME imath.txt)
  endfunction()
  add_foliage_alembic()
endif()
if(FOLIAGE_BUNDLED_PNG)
  enable_language(C)
  set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
  set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
  set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(zlib URL https://codeload.github.com/madler/zlib/tar.gz/refs/tags/v1.3.2 URL_HASH SHA256=b99a0b86c0ba9360ec7e78c4f1e43b1cbdf1e6936c8fa0f6835c0cd694a495a1 OVERRIDE_FIND_PACKAGE)
  FetchContent_MakeAvailable(zlib)
  add_library(ZLIB::ZLIB ALIAS zlibstatic)
  set(ZLIB_INCLUDE_DIRS "${zlib_BINARY_DIR};${zlib_SOURCE_DIR}")
  set(PNG_HARDWARE_OPTIMIZATIONS OFF CACHE BOOL "" FORCE)
  set(PNG_SHARED OFF CACHE BOOL "" FORCE)
  set(PNG_FRAMEWORK OFF CACHE BOOL "" FORCE)
  set(PNG_STATIC ON CACHE BOOL "" FORCE)
  set(PNG_TESTS OFF CACHE BOOL "" FORCE)
  set(PNG_TOOLS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(png URL https://codeload.github.com/pnggroup/libpng/tar.gz/refs/tags/v1.6.58 URL_HASH SHA256=a9d4df463d36a6e5f9c29bd6f4967312d17e996c1854f3511f833924eb1993cf)
  FetchContent_MakeAvailable(png)
  set(FOLIAGE_PNG png_static)
else()
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(SYSTEM_PNG QUIET IMPORTED_TARGET libpng)
  endif()
  if(TARGET PkgConfig::SYSTEM_PNG)
    set(FOLIAGE_PNG PkgConfig::SYSTEM_PNG)
  else()
    find_package(PNG REQUIRED)
    set(FOLIAGE_PNG PNG::PNG)
  endif()
endif()
